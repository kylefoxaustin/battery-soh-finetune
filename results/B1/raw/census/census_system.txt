### 2. anonymous mapping test (MAP_POPULATE, then MAP_HUGETLB), 512 MB and 2 GB; released
$ timeout -k 10 120 python3 - <<'EOF'
import mmap, time
for mb in (512, 2048):
    for lab, fl in (('POPULATE', mmap.MAP_PRIVATE | mmap.MAP_ANONYMOUS | getattr(mmap, 'MAP_POPULATE', 0x8000)),
                    ('HUGETLB+POPULATE', mmap.MAP_PRIVATE | mmap.MAP_ANONYMOUS | 0x40000 | getattr(mmap, 'MAP_POPULATE', 0x8000))):
        t = time.perf_counter()
        try:
            m = mmap.mmap(-1, mb << 20, flags=fl)
            m[0:1] = b'x'; m[-1:] = b'x'
            print(f'{mb} MB {lab}: OK in {time.perf_counter() - t:.2f} s'); m.close()
        except OSError as e:
            print(f'{mb} MB {lab}: FAILED ({e})')
EOF
512 MB POPULATE: OK in 0.04 s
512 MB HUGETLB+POPULATE: FAILED ([Errno 12] Cannot allocate memory)
2048 MB POPULATE: OK in 0.18 s
2048 MB HUGETLB+POPULATE: FAILED ([Errno 12] Cannot allocate memory)
